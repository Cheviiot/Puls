cmake_minimum_required(VERSION 3.21)
project(Puls VERSION 0.4.0 LANGUAGES CXX)

option(PULS_BUILD_GUI "Build the Qt Widgets frontend" ON)
option(PULS_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(PULS_DEPLOY_RUNTIME "Install a relocatable native desktop Qt runtime" OFF)
set(PULS_VERSION_LABEL "${PROJECT_VERSION}-cpp-dev" CACHE STRING "Public CLI and artifact version label")
include(CTest)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOGEN_PARALLEL 1)
if(MSVC)
    add_compile_options(/utf-8)
endif()
if(PULS_DEPLOY_RUNTIME AND UNIX AND NOT APPLE AND NOT ANDROID)
    find_program(PULS_PATCHELF_EXECUTABLE patchelf REQUIRED)
    # System SDK plugins often have no spare RPATH entry for CMake to replace.
    # Set this before finding Core, which generates the Qt deployment support.
    set(QT_DEPLOY_USE_PATCHELF ON)
    set(QT_DEPLOY_PATCHELF_EXECUTABLE "${PULS_PATCHELF_EXECUTABLE}")
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network WebSockets)
if(PULS_BUILD_GUI)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
endif()

add_library(puls_core STATIC cpp/src/core.cpp cpp/src/backend.cpp cpp/src/transfer.cpp
    cpp/src/yandex_throughput.cpp cpp/src/speedtest_throughput.cpp)
target_include_directories(puls_core PUBLIC cpp/include)
target_link_libraries(puls_core PUBLIC Qt6::Core Qt6::Network Qt6::WebSockets)
add_library(puls_cli STATIC cpp/src/cli.cpp)
target_link_libraries(puls_cli PUBLIC puls_core)
target_compile_definitions(puls_cli PRIVATE PULS_VERSION="${PULS_VERSION_LABEL}")
qt_add_executable(puls cpp/src/main.cpp)
target_link_libraries(puls PRIVATE puls_cli)
if(WIN32)
    enable_language(RC)
    set(PULS_WINDOWS_ICON "${PROJECT_SOURCE_DIR}/internal/gui/assets/Icon.ico")
    configure_file(cpp/puls.rc.in "${PROJECT_BINARY_DIR}/puls.rc" @ONLY)
    target_sources(puls PRIVATE "${PROJECT_BINARY_DIR}/puls.rc")
endif()
if(PULS_BUILD_GUI)
    add_library(puls_gui STATIC cpp/src/gui.cpp)
    target_link_libraries(puls_gui PUBLIC puls_cli Qt6::Widgets)
    target_sources(puls PRIVATE cpp/resources.qrc)
    target_link_libraries(puls PRIVATE puls_gui)
    target_compile_definitions(puls PRIVATE PULS_BUILD_GUI=1)
    set_target_properties(puls PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.cheviiot.puls"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
endif()
if(ANDROID AND NOT PULS_BUILD_GUI)
    message(FATAL_ERROR "Android requires PULS_BUILD_GUI=ON")
endif()
if(ANDROID)
    if(Qt6_VERSION VERSION_LESS 6.8)
        message(FATAL_ERROR "Android packaging requires Qt 6.8 or later")
    endif()
    set(PULS_ANDROID_VERSION_CODE "4000" CACHE STRING "Monotonic Android version code")
    if(NOT PULS_ANDROID_VERSION_CODE MATCHES "^[1-9][0-9]*$" OR
       PULS_ANDROID_VERSION_CODE GREATER 2100000000)
        message(FATAL_ERROR "PULS_ANDROID_VERSION_CODE must be an integer from 1 to 2100000000")
    endif()
    if(NOT PULS_VERSION_LABEL MATCHES "^[0-9A-Za-z.+_-]+$")
        message(FATAL_ERROR "Android PULS_VERSION_LABEL must be a version string")
    endif()
    set(PULS_ANDROID_OPENSSL_ROOT "" CACHE PATH "OpenSSL 3 runtime directory containing one subdirectory per Android ABI")
    if(NOT IS_DIRECTORY "${PULS_ANDROID_OPENSSL_ROOT}")
        message(FATAL_ERROR "Android requires PULS_ANDROID_OPENSSL_ROOT with packaged OpenSSL 3 libraries")
    endif()
    set(puls_android_abis "${CMAKE_ANDROID_ARCH_ABI}" ${QT_ANDROID_ABIS})
    list(REMOVE_DUPLICATES puls_android_abis)
    set(puls_android_ssl_libraries "")
    foreach(abi IN LISTS puls_android_abis)
        foreach(library IN ITEMS crypto ssl)
            set(path "${PULS_ANDROID_OPENSSL_ROOT}/${abi}/lib${library}_3.so")
            if(NOT EXISTS "${path}")
                message(FATAL_ERROR "Missing Android TLS runtime: ${path}")
            endif()
            list(APPEND puls_android_ssl_libraries "${path}")
        endforeach()
    endforeach()
    if(NOT EXISTS "${PULS_ANDROID_OPENSSL_ROOT}/LICENSE.txt")
        message(FATAL_ERROR "Android OpenSSL runtime must include LICENSE.txt")
    endif()
    # The additional ABI projects do not inherit arbitrary cache variables.
    list(APPEND QT_ANDROID_MULTI_ABI_FORWARD_VARS PULS_ANDROID_OPENSSL_ROOT
        PULS_ANDROID_VERSION_CODE PULS_VERSION_LABEL PULS_BUILD_GUI PULS_DEPLOY_RUNTIME
        BUILD_TESTING)
    list(REMOVE_DUPLICATES QT_ANDROID_MULTI_ABI_FORWARD_VARS)
    set(puls_android_package "${PROJECT_BINARY_DIR}/puls-android-package")
    file(MAKE_DIRECTORY "${puls_android_package}/res/drawable"
        "${puls_android_package}/res/raw")
    configure_file(cpp/android/AndroidManifest.xml.in
        "${puls_android_package}/AndroidManifest.xml" @ONLY)
    configure_file(internal/gui/assets/Icon.png
        "${puls_android_package}/res/drawable/puls_icon.png" COPYONLY)
    configure_file("${PULS_ANDROID_OPENSSL_ROOT}/LICENSE.txt"
        "${puls_android_package}/res/raw/openssl_license.txt" COPYONLY)
    configure_file(LICENSE "${puls_android_package}/res/raw/puls_license.txt" COPYONLY)
    file(GLOB puls_android_qt_notices CONFIGURE_DEPENDS
        "${PROJECT_SOURCE_DIR}/cmake/qt-licenses/*.txt")
    foreach(notice IN LISTS puls_android_qt_notices)
        get_filename_component(name "${notice}" NAME_WLE)
        string(TOLOWER "${name}" name)
        string(REGEX REPLACE "[^a-z0-9_]" "_" name "${name}")
        configure_file("${notice}" "${puls_android_package}/res/raw/qt_${name}.txt" COPYONLY)
    endforeach()
    set_target_properties(puls PROPERTIES
        QT_ANDROID_PACKAGE_NAME "io.github.cheviiot.puls"
        QT_ANDROID_APP_NAME "Puls"
        QT_ANDROID_PACKAGE_SOURCE_DIR "${puls_android_package}"
        QT_ANDROID_VERSION_NAME "${PULS_VERSION_LABEL}"
        QT_ANDROID_VERSION_CODE "${PULS_ANDROID_VERSION_CODE}"
        QT_ANDROID_MIN_SDK_VERSION 28
        QT_ANDROID_TARGET_SDK_VERSION 35
        QT_ANDROID_SDK_BUILD_TOOLS_REVISION "35.0.0"
        QT_ANDROID_EXTRA_LIBS "${puls_android_ssl_libraries}")
    target_link_options(puls PRIVATE "-Wl,-z,max-page-size=16384")
endif()

set(PULS_TARGETS puls_core puls_cli puls)
if(PULS_BUILD_GUI)
    list(APPEND PULS_TARGETS puls_gui)
endif()
foreach(target IN LISTS PULS_TARGETS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(PULS_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    foreach(target IN LISTS PULS_TARGETS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(puls PRIVATE -fsanitize=address,undefined)
endif()

if(BUILD_TESTING AND NOT ANDROID)
    add_executable(puls_core_tests cpp/tests/core_tests.cpp)
    target_link_libraries(puls_core_tests PRIVATE puls_cli)
    add_test(NAME core_contract COMMAND puls_core_tests)
    add_executable(puls_transfer_tests cpp/tests/transfer_tests.cpp)
    target_link_libraries(puls_transfer_tests PRIVATE puls_core)
    add_test(NAME transfer_contract COMMAND puls_transfer_tests)
    add_executable(puls_yandex_transfer_tests cpp/tests/yandex_transfer_tests.cpp)
    target_link_libraries(puls_yandex_transfer_tests PRIVATE puls_core)
    add_test(NAME yandex_transfer_contract COMMAND puls_yandex_transfer_tests)
    add_executable(puls_speedtest_transfer_tests cpp/tests/speedtest_transfer_tests.cpp)
    target_link_libraries(puls_speedtest_transfer_tests PRIVATE puls_core)
    add_test(NAME speedtest_transfer_contract COMMAND puls_speedtest_transfer_tests)
    add_executable(puls_fallback_tests cpp/tests/fallback_tests.cpp)
    target_link_libraries(puls_fallback_tests PRIVATE puls_core)
    add_test(NAME fallback_contract COMMAND puls_fallback_tests)
    add_test(NAME cli_help COMMAND puls help)
    add_test(NAME cli_version COMMAND puls version)
    add_executable(puls_cli_process_tests cpp/tests/cli_process_tests.cpp)
    target_link_libraries(puls_cli_process_tests PRIVATE puls_cli)
    target_compile_definitions(puls_cli_process_tests PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/cpp/tests")
    add_test(NAME cli_process_contract COMMAND puls_cli_process_tests $<TARGET_FILE:puls>)
    if(PULS_BUILD_GUI)
        add_executable(puls_gui_tests cpp/tests/gui_tests.cpp)
        target_link_libraries(puls_gui_tests PRIVATE puls_gui)
        add_test(NAME gui_contract COMMAND puls_gui_tests)
        set_tests_properties(gui_contract PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()

install(TARGETS puls BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(cmake/PulsDeployment.cmake)
